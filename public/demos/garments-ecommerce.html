Use this lightweight async function in any React, Next.js, or HTML checkout: 

async function pushOrderToBitrix24(order) {
  const webhookUrl = 'https://YOUR_SUBDOMAIN.bitrix24.in/rest/1/TOKEN/crm.deal.add.json';
  
  const payload = {
    fields: {
      TITLE: `Garment Order #${order.id} - ${order.custName}`,
      STAGE_ID: 'NEW',
      OPPORTUNITY: order.amount,
      CURRENCY_ID: 'INR',
      COMMENTS: `Items: ${order.items}\nPhone: ${order.custPhone}\nAddress: ${order.address}\nPayment: ${order.method}`
    }
  };

  const res = await fetch(webhookUrl, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload)
  });
  return await res.json();
}
